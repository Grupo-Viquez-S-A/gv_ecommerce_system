import { createClient } from "npm:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers":
    "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

const ADMIN_DEPARTMENT_NAMES = ["INFORMATICA"];
const ADMIN_ROLE_CODES = ["ENCARGADO"];
const ADMIN_ROLE_NAMES = ["ENCARGADO"];

type ProfilePayload = {
  name?: unknown;
  surname?: unknown;
  identification?: unknown;
  phone?: unknown;
};

type MembershipPayload = {
  membership_id?: unknown;
  company_id?: unknown;
  department_id?: unknown;
  role_id?: unknown;
};

type DepartmentPayload = {
  department_id?: unknown;
  name?: unknown;
  email?: unknown;
  is_active?: unknown;
};

type RolePayload = {
  role_id?: unknown;
  role_name?: unknown;
  role_code?: unknown;
  description?: unknown;
  is_active?: unknown;
};

type AdminSettingsRequest = {
  action?: unknown;
  email?: unknown;
  password?: unknown;
  userId?: unknown;
  isActive?: unknown;
  departmentId?: unknown;
  roleId?: unknown;

  /*
    Recibe los IDs de los módulos seleccionados
    desde los checkboxes del modal de usuario.
  */
  moduleIds?: unknown;

  profile?: ProfilePayload;
  membership?: MembershipPayload;
  department?: DepartmentPayload;
  role?: RolePayload;
};

function jsonResponse(body: unknown, status = 200) {
  return new Response(JSON.stringify(body), {
    status,
    headers: {
      ...corsHeaders,
      "Content-Type": "application/json",
    },
  });
}

function getErrorMessage(
  error: unknown,
  fallback = "Ocurrió un error inesperado."
) {
  if (typeof error === "string" && error.trim()) {
    return error;
  }

  if (error instanceof Error && error.message) {
    return error.message;
  }

  if (
    typeof error === "object" &&
    error !== null &&
    "message" in error &&
    typeof (error as { message?: unknown }).message === "string"
  ) {
    return (error as { message: string }).message;
  }

  try {
    const serialized = JSON.stringify(error);

    if (serialized && serialized !== "{}") {
      return serialized;
    }
  } catch {
    // Mantiene el fallback.
  }

  return fallback;
}

function createHttpError(message: unknown, status = 400) {
  const error = new Error(getErrorMessage(message)) as Error & {
    status?: number;
  };

  error.status = status;

  return error;
}

function getRelationValue(value: unknown) {
  if (Array.isArray(value)) {
    return value[0] ?? null;
  }

  return value;
}

function getRequiredString(value: unknown) {
  return typeof value === "string" ? value.trim() : "";
}

function getOptionalString(value: unknown) {
  const result = getRequiredString(value);

  return result || null;
}

function getBoolean(value: unknown, defaultValue = true) {
  return typeof value === "boolean" ? value : defaultValue;
}

function getStringArray(value: unknown) {
  if (!Array.isArray(value)) {
    return [];
  }

  return [
    ...new Set(
      value
        .filter(
          (item): item is string =>
            typeof item === "string" && item.trim().length > 0
        )
        .map((item) => item.trim())
    ),
  ];
}

function normalizeAuthorizationValue(value: unknown) {
  return String(value ?? "")
    .trim()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toUpperCase();
}

function getToday() {
  return new Date().toISOString().split("T")[0];
}

function getNow() {
  return new Date().toISOString();
}

/*
  Reemplaza por completo los permisos de módulos de un usuario.

  1. Verifica que los módulos seleccionados estén activos.
  2. Elimina los permisos anteriores del usuario.
  3. Inserta los módulos seleccionados con permisos completos.

  La relación correcta en esta base es:

  profiles.user_id
      ↓
  profile_modules.profile_id
      ↓
  modules.module_id
*/
async function replaceUserModuleAccess(
  supabaseAdmin: any,
  profileId: string,
  moduleIds: string[]
) {
  const uniqueModuleIds = getStringArray(moduleIds);

  /*
    profile_modules guarda únicamente módulos adicionales:
    - Nunca módulos públicos.
    - Nunca módulos padre no asignables.
    - Nunca módulos desactivados.
  */
  const { data: assignableModules, error: modulesError } =
    await supabaseAdmin
      .from("modules")
      .select("module_id")
      .eq("is_active", true)
      .eq("is_public", false)
      .eq("is_assignable", true);

  if (modulesError) {
    throw new Error(
      modulesError.message ||
        "No fue posible validar los módulos disponibles."
    );
  }

  const validModuleIds = new Set(
    (assignableModules || []).map(
      (module: { module_id: string }) => module.module_id
    )
  );

  const invalidModuleIds = uniqueModuleIds.filter(
    (moduleId) => !validModuleIds.has(moduleId)
  );

  if (invalidModuleIds.length > 0) {
    throw new Error(
      "Se intentó asignar un módulo inválido, público, no asignable o inactivo."
    );
  }

  /*
    Reemplaza los permisos individuales anteriores.
    Los permisos automáticos por departamento se almacenan
    en department_modules y no se modifican aquí.
  */
  const { error: deleteError } = await supabaseAdmin
    .from("profile_modules")
    .delete()
    .eq("profile_id", profileId);

  if (deleteError) {
    throw new Error(
      deleteError.message ||
        "No fue posible eliminar los permisos adicionales anteriores."
    );
  }

  if (uniqueModuleIds.length === 0) {
    return;
  }

  const now = getNow();

  const assignments = uniqueModuleIds.map((moduleId) => ({
    profile_id: profileId,
    module_id: moduleId,
    can_view: true,
    can_create: true,
    can_edit: true,
    can_delete: true,
    created_at: now,
    updated_at: now,
  }));

  const { error: insertError } = await supabaseAdmin
    .from("profile_modules")
    .insert(assignments);

  if (insertError) {
    throw new Error(
      insertError.message ||
        "No fue posible guardar los permisos adicionales."
    );
  }
}

async function validateAdministrator(
  request: Request,
  supabaseAdmin: ReturnType<typeof createClient>
) {
  const authorization = request.headers.get("Authorization");

  if (!authorization?.startsWith("Bearer ")) {
    throw createHttpError("No se encontró una sesión válida.", 401);
  }

  const token = authorization.replace("Bearer ", "");

  const {
    data: { user },
    error: userError,
  } = await supabaseAdmin.auth.getUser(token);

  if (userError || !user) {
    throw createHttpError("La sesión no es válida o expiró.", 401);
  }

  const { data: memberships, error: membershipError } =
    await supabaseAdmin
      .from("user_memberships")
      .select(`
        roles!inner(role_name, role_code),
        departments!inner(name)
      `)
      .eq("user_id", user.id)
      .eq("is_active", true);

  if (membershipError) {
    throw createHttpError(
      "No fue posible validar los permisos administrativos.",
      403
    );
  }

  const hasAdministrationAccess = (memberships ?? []).some(
    (membership) => {
      const role = getRelationValue(membership.roles) as {
        role_name?: string;
        role_code?: string;
      } | null;
      const department = getRelationValue(membership.departments) as {
        name?: string;
      } | null;

      const normalizedRoleCode = normalizeAuthorizationValue(
        role?.role_code
      );
      const normalizedRoleName = normalizeAuthorizationValue(
        role?.role_name
      );
      const normalizedDepartmentName = normalizeAuthorizationValue(
        department?.name
      );

      const isInformationTechnology =
        ADMIN_DEPARTMENT_NAMES.includes(normalizedDepartmentName);

      const isDepartmentLead =
        ADMIN_ROLE_CODES.includes(normalizedRoleCode) ||
        ADMIN_ROLE_NAMES.includes(normalizedRoleName);

      return isInformationTechnology && isDepartmentLead;
    }
  );

  if (!hasAdministrationAccess) {
    throw createHttpError(
      "Solo usuarios del departamento de Informática con rol Encargado pueden administrar usuarios y configuración.",
      403
    );
  }

  return user;
}

Deno.serve(async (request: Request) => {
  if (request.method === "OPTIONS") {
    return new Response("ok", {
      headers: corsHeaders,
    });
  }

  if (request.method !== "POST") {
    return jsonResponse(
      {
        ok: false,
        error: "Método no permitido.",
      },
      405
    );
  }

  try {
    const supabaseUrl = Deno.env.get("SUPABASE_URL");
    const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

    if (!supabaseUrl || !serviceRoleKey) {
      throw createHttpError(
        "No se encontraron las credenciales internas de Supabase.",
        500
      );
    }

    const supabaseAdmin = createClient(supabaseUrl, serviceRoleKey, {
      auth: {
        autoRefreshToken: false,
        persistSession: false,
      },
    });

    const currentUser = await validateAdministrator(
      request,
      supabaseAdmin
    );

    const body = (await request.json()) as AdminSettingsRequest;
    const action = getRequiredString(body?.action);

    // ==========================================================
    // LISTAR USUARIOS, DEPARTAMENTOS, ROLES, EMPRESAS Y MÓDULOS
    // ==========================================================
    if (action === "list") {
      const [
        usersResponse,
        departmentsResponse,
        rolesResponse,
        companiesResponse,
        modulesResponse,
        departmentModulesResponse,
      ] = await Promise.all([
        supabaseAdmin
          .from("profiles")
          .select(`
            user_id,
            name,
            surname,
            email,
            identification,
            phone,
            is_active,
            created_at,

            user_memberships (
              membership_id,
              company_id,
              department_id,
              role_id,
              is_active,
              start_date,
              end_date,

              companies (
                company_id,
                company_name
              ),

              departments (
                department_id,
                name
              ),

              roles (
                role_id,
                role_name,
                role_code
              )
            ),

            profile_modules (
              profile_id,
              module_id,
              can_view,
              can_create,
              can_edit,
              can_delete,
              created_at,
              updated_at
            )
          `)
          .order("created_at", { ascending: false }),

        supabaseAdmin
          .from("departments")
          .select("*")
          .order("name", { ascending: true }),

        supabaseAdmin
          .from("roles")
          .select("*")
          .order("role_name", { ascending: true }),

        supabaseAdmin
          .from("companies")
          .select("company_id, company_name, is_active")
          .order("company_name", { ascending: true }),

        supabaseAdmin
          .from("modules")
          .select(`
            module_id,
            module_code,
            name,
            parent_module_id,
            route,
            display_order,
            is_public,
            is_assignable,
            is_active
          `)
          .eq("is_active", true)
          .order("display_order", { ascending: true }),

        supabaseAdmin
          .from("department_modules")
          .select(`
            department_id,
            module_id,
            can_view,
            can_create,
            can_edit,
            can_delete
          `),
      ]);

      const firstError = [
        usersResponse.error,
        departmentsResponse.error,
        rolesResponse.error,
        companiesResponse.error,
        modulesResponse.error,
        departmentModulesResponse.error,
      ].find(Boolean);

      if (firstError) {
        throw createHttpError(firstError);
      }

      return jsonResponse({
        ok: true,
        users: usersResponse.data ?? [],
        departments: departmentsResponse.data ?? [],
        roles: rolesResponse.data ?? [],
        companies: companiesResponse.data ?? [],
        modules: modulesResponse.data ?? [],
        departmentModules: departmentModulesResponse.data ?? [],
      });
    }

    // ==========================================================
    // CREAR USUARIO
    // ==========================================================
    if (action === "create-user") {
      const email = getRequiredString(body.email).toLowerCase();
      const password = getRequiredString(body.password);
      const profile = body.profile ?? {};
      const membership = body.membership ?? {};
      const moduleIds = getStringArray(body.moduleIds);

      const name = getRequiredString(profile.name);
      const surname = getRequiredString(profile.surname);

      const companyId = getRequiredString(membership.company_id);
      const departmentId = getRequiredString(membership.department_id);
      const roleId = getRequiredString(membership.role_id);

      const missingFields: string[] = [];

      if (!email) missingFields.push("correo electrónico");
      if (!password) missingFields.push("contraseña temporal");
      if (!name) missingFields.push("nombre");
      if (!surname) missingFields.push("apellidos");
      if (!companyId) missingFields.push("empresa");
      if (!departmentId) missingFields.push("departamento");
      if (!roleId) missingFields.push("rol");

      console.log(
        "DIAGNOSTICO CREATE USER:",
        JSON.stringify({
          hasEmail: Boolean(email),
          passwordLength: password.length,
          hasName: Boolean(name),
          hasSurname: Boolean(surname),
          hasCompany: Boolean(companyId),
          hasDepartment: Boolean(departmentId),
          hasRole: Boolean(roleId),
          totalModules: moduleIds.length,
          missingFields,
        })
      );

      if (missingFields.length > 0) {
        return jsonResponse(
          {
            ok: false,
            error: `Faltan o son inválidos estos datos: ${missingFields.join(
              ", "
            )}.`,
          },
          400
        );
      }

      if (password.length < 8) {
        return jsonResponse(
          {
            ok: false,
            error:
              "La contraseña temporal debe tener al menos 8 caracteres.",
          },
          400
        );
      }

      const { data: authData, error: authError } =
        await supabaseAdmin.auth.admin.createUser({
          email,
          password,
          email_confirm: true,
          user_metadata: {
            name,
            surname,
          },
        });

      if (authError || !authData.user) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(
              authError,
              "No fue posible crear la cuenta de acceso."
            ),
          },
          400
        );
      }

      const newUserId = authData.user.id;

      const { error: profileError } = await supabaseAdmin
        .from("profiles")
        .upsert(
          {
            user_id: newUserId,
            name,
            surname,
            email,
            identification: getOptionalString(profile.identification),
            phone: getOptionalString(profile.phone),
            is_active: true,
            updated_at: getNow(),
          },
          {
            onConflict: "user_id",
          }
        );

      if (profileError) {
        await supabaseAdmin.auth.admin.deleteUser(newUserId);

        return jsonResponse(
          {
            ok: false,
            error: `La cuenta fue creada, pero falló al guardar el perfil: ${getErrorMessage(
              profileError
            )}`,
          },
          400
        );
      }

      const { error: membershipError } = await supabaseAdmin
        .from("user_memberships")
        .insert({
          user_id: newUserId,
          company_id: companyId,
          department_id: departmentId,
          role_id: roleId,
          is_active: true,
          start_date: getToday(),
          created_at: getNow(),
          updated_at: getNow(),
        });

      if (membershipError) {
        await supabaseAdmin.auth.admin.deleteUser(newUserId);

        return jsonResponse(
          {
            ok: false,
            error: `La cuenta fue creada, pero falló al asignar empresa, departamento o rol: ${getErrorMessage(
              membershipError
            )}`,
          },
          400
        );
      }

      /*
        Guarda los módulos seleccionados por los checkboxes.
        profile_modules.profile_id usa el mismo UUID de profiles.user_id.
      */
      try {
        await replaceUserModuleAccess(
          supabaseAdmin,
          newUserId,
          moduleIds
        );
      } catch (moduleAccessError) {
        await supabaseAdmin.auth.admin.deleteUser(newUserId);

        return jsonResponse(
          {
            ok: false,
            error: `La cuenta fue creada, pero no fue posible asignar los módulos: ${getErrorMessage(
              moduleAccessError
            )}`,
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: "Usuario creado correctamente.",
      });
    }

    // ==========================================================
    // EDITAR USUARIO
    // ==========================================================
    if (action === "update-user") {
      const userId = getRequiredString(body.userId);
      const email = getRequiredString(body.email).toLowerCase();
      const password = getRequiredString(body.password);
      const profile = body.profile ?? {};
      const membership = body.membership ?? {};
      const moduleIds = getStringArray(body.moduleIds);

      const name = getRequiredString(profile.name);
      const surname = getRequiredString(profile.surname);

      const companyId = getRequiredString(membership.company_id);
      const departmentId = getRequiredString(membership.department_id);
      const roleId = getRequiredString(membership.role_id);
      const membershipId = getRequiredString(membership.membership_id);

      const missingFields: string[] = [];

      if (!userId) missingFields.push("identificador del usuario");
      if (!email) missingFields.push("correo electrónico");
      if (!name) missingFields.push("nombre");
      if (!surname) missingFields.push("apellidos");
      if (!companyId) missingFields.push("empresa");
      if (!departmentId) missingFields.push("departamento");
      if (!roleId) missingFields.push("rol");

      if (missingFields.length > 0) {
        return jsonResponse(
          {
            ok: false,
            error: `Faltan o son inválidos estos datos: ${missingFields.join(
              ", "
            )}.`,
          },
          400
        );
      }

      if (password && password.length < 8) {
        return jsonResponse(
          {
            ok: false,
            error:
              "La nueva contraseña debe tener al menos 8 caracteres.",
          },
          400
        );
      }

      const authPayload: {
        email: string;
        email_confirm: boolean;
        password?: string;
        user_metadata: {
          name: string;
          surname: string;
        };
      } = {
        email,
        email_confirm: true,
        user_metadata: {
          name,
          surname,
        },
      };

      if (password) {
        authPayload.password = password;
      }

      const { error: authError } =
        await supabaseAdmin.auth.admin.updateUserById(
          userId,
          authPayload
        );

      if (authError) {
        return jsonResponse(
          {
            ok: false,
            error: `No fue posible actualizar la cuenta de acceso: ${getErrorMessage(
              authError
            )}`,
          },
          400
        );
      }

      const { error: profileError } = await supabaseAdmin
        .from("profiles")
        .upsert(
          {
            user_id: userId,
            name,
            surname,
            email,
            identification: getOptionalString(profile.identification),
            phone: getOptionalString(profile.phone),
            updated_at: getNow(),
          },
          {
            onConflict: "user_id",
          }
        );

      if (profileError) {
        return jsonResponse(
          {
            ok: false,
            error: `No fue posible actualizar el perfil: ${getErrorMessage(
              profileError
            )}`,
          },
          400
        );
      }

      const membershipData = {
        company_id: companyId,
        department_id: departmentId,
        role_id: roleId,
        updated_at: getNow(),
      };

      let membershipError = null;

      if (membershipId) {
        const response = await supabaseAdmin
          .from("user_memberships")
          .update(membershipData)
          .eq("membership_id", membershipId)
          .eq("user_id", userId);

        membershipError = response.error;
      } else {
        const response = await supabaseAdmin
          .from("user_memberships")
          .insert({
            user_id: userId,
            ...membershipData,
            is_active: true,
            start_date: getToday(),
            created_at: getNow(),
          });

        membershipError = response.error;
      }

      if (membershipError) {
        return jsonResponse(
          {
            ok: false,
            error: `No fue posible actualizar la asignación del usuario: ${getErrorMessage(
              membershipError
            )}`,
          },
          400
        );
      }

      /*
        Reemplaza los módulos anteriores por los actualmente marcados
        en el modal de edición.
      */
      try {
        await replaceUserModuleAccess(
          supabaseAdmin,
          userId,
          moduleIds
        );
      } catch (moduleAccessError) {
        return jsonResponse(
          {
            ok: false,
            error: `El usuario fue actualizado, pero no fue posible guardar los módulos: ${getErrorMessage(
              moduleAccessError
            )}`,
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: "Usuario actualizado correctamente.",
      });
    }

    // ==========================================================
    // ACTIVAR O DESACTIVAR USUARIO
    // ==========================================================
    if (action === "set-user-status") {
      const userId = getRequiredString(body.userId);
      const isActive = body.isActive;

      if (!userId || typeof isActive !== "boolean") {
        return jsonResponse(
          {
            ok: false,
            error: "Datos de usuario inválidos.",
          },
          400
        );
      }

      if (userId === currentUser.id && !isActive) {
        return jsonResponse(
          {
            ok: false,
            error: "No puedes desactivar tu propia cuenta.",
          },
          400
        );
      }

      const now = getNow();

      const { error: profileError } = await supabaseAdmin
        .from("profiles")
        .update({
          is_active: isActive,
          updated_at: now,
        })
        .eq("user_id", userId);

      if (profileError) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(profileError),
          },
          400
        );
      }

      const { error: membershipError } = await supabaseAdmin
        .from("user_memberships")
        .update({
          is_active: isActive,
          end_date: isActive ? null : getToday(),
          updated_at: now,
        })
        .eq("user_id", userId);

      if (membershipError) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(membershipError),
          },
          400
        );
      }

      const { error: authError } =
        await supabaseAdmin.auth.admin.updateUserById(userId, {
          ban_duration: isActive ? "none" : "876000h",
        });

      if (authError) {
        return jsonResponse(
          {
            ok: false,
            error: `No fue posible actualizar el acceso del usuario: ${getErrorMessage(
              authError
            )}`,
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: `Usuario ${
          isActive ? "activado" : "desactivado"
        } correctamente.`,
      });
    }

    // ==========================================================
    // CREAR O EDITAR DEPARTAMENTO
    // ==========================================================
    if (action === "save-department") {
      const department = body.department ?? {};
      const departmentId = getRequiredString(department.department_id);
      const name = getRequiredString(department.name);
      const email = getOptionalString(department.email);
      const isActive = getBoolean(department.is_active, true);

      if (!name) {
        return jsonResponse(
          {
            ok: false,
            error: "El nombre del departamento es obligatorio.",
          },
          400
        );
      }

      const departmentData = {
        name,
        email,
        is_active: isActive,
        updated_at: getNow(),
      };

      let responseError = null;

      if (departmentId) {
        const response = await supabaseAdmin
          .from("departments")
          .update(departmentData)
          .eq("department_id", departmentId);

        responseError = response.error;
      } else {
        const response = await supabaseAdmin
          .from("departments")
          .insert({
            ...departmentData,
            created_at: getNow(),
          });

        responseError = response.error;
      }

      if (responseError) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(responseError),
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: "Departamento guardado correctamente.",
      });
    }

    // ==========================================================
    // ACTIVAR O DESACTIVAR DEPARTAMENTO
    // ==========================================================
    if (action === "set-department-status") {
      const departmentId = getRequiredString(body.departmentId);
      const isActive = body.isActive;

      if (!departmentId || typeof isActive !== "boolean") {
        return jsonResponse(
          {
            ok: false,
            error: "Datos de departamento inválidos.",
          },
          400
        );
      }

      const { error } = await supabaseAdmin
        .from("departments")
        .update({
          is_active: isActive,
          updated_at: getNow(),
        })
        .eq("department_id", departmentId);

      if (error) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(error),
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: `Departamento ${
          isActive ? "activado" : "desactivado"
        } correctamente.`,
      });
    }

    // ==========================================================
    // CREAR O EDITAR ROL
    // ==========================================================
    if (action === "save-role") {
      const role = body.role ?? {};
      const roleId = getRequiredString(role.role_id);
      const roleName = getRequiredString(role.role_name);
      const roleCode = getRequiredString(role.role_code).toUpperCase();
      const description = getOptionalString(role.description);
      const isActive = getBoolean(role.is_active, true);

      if (!roleName || !roleCode) {
        return jsonResponse(
          {
            ok: false,
            error: "El nombre y código del rol son obligatorios.",
          },
          400
        );
      }

      const roleData = {
        role_name: roleName,
        role_code: roleCode,
        description,
        is_active: isActive,
        updated_at: getNow(),
      };

      let responseError = null;

      if (roleId) {
        const response = await supabaseAdmin
          .from("roles")
          .update(roleData)
          .eq("role_id", roleId);

        responseError = response.error;
      } else {
        const response = await supabaseAdmin
          .from("roles")
          .insert({
            ...roleData,
            created_at: getNow(),
          });

        responseError = response.error;
      }

      if (responseError) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(responseError),
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: "Rol guardado correctamente.",
      });
    }

    // ==========================================================
    // ACTIVAR O DESACTIVAR ROL
    // ==========================================================
    if (action === "set-role-status") {
      const roleId = getRequiredString(body.roleId);
      const isActive = body.isActive;

      if (!roleId || typeof isActive !== "boolean") {
        return jsonResponse(
          {
            ok: false,
            error: "Datos de rol inválidos.",
          },
          400
        );
      }

      const { error } = await supabaseAdmin
        .from("roles")
        .update({
          is_active: isActive,
          updated_at: getNow(),
        })
        .eq("role_id", roleId);

      if (error) {
        return jsonResponse(
          {
            ok: false,
            error: getErrorMessage(error),
          },
          400
        );
      }

      return jsonResponse({
        ok: true,
        message: `Rol ${
          isActive ? "activado" : "desactivado"
        } correctamente.`,
      });
    }

    return jsonResponse(
      {
        ok: false,
        error: "Acción no reconocida.",
      },
      400
    );
  } catch (error) {
    console.error("ADMIN SETTINGS ERROR:", error);

    const status =
      typeof error === "object" &&
      error !== null &&
      "status" in error &&
      typeof (error as { status?: unknown }).status === "number"
        ? (error as { status: number }).status
        : 500;

    return jsonResponse(
      {
        ok: false,
        error: getErrorMessage(error),
      },
      status
    );
  }
});
